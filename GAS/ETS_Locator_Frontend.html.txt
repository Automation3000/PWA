    <script>
        // REPLACE WITH YOUR PUBLISHED WEB APP URL:
        const GAS_API_URL = "YOUR_WEB_APP_URL_HERE";
        
        let etsList = [];

        function initApp() {
            // Theme setup
            if (localStorage.getItem('tabreed_theme') === 'dark') {
                document.body.classList.add('dark-mode');
            }
            
            refreshTable(false); // Initial load without pop animation
        }

        function refreshTable(showAnim = true) {
            if(showAnim) {
                const btn = document.getElementById('btn-refresh');
                if (btn) {
                    btn.classList.add('pop-anim');
                    setTimeout(() => btn.classList.remove('pop-anim'), 350);
                }
            }
            
            document.getElementById('etsTableBody').innerHTML = `<tr><td colspan="3" style="text-align:center;padding:30px;color:var(--text-muted);"><i class="fas fa-circle-notch fa-spin"></i> Loading Data from Cloud...</td></tr>`;

            if(GAS_API_URL === "YOUR_WEB_APP_URL_HERE") {
                document.getElementById('etsTableBody').innerHTML = `<tr><td colspan="3" style="text-align:center;padding:30px;color:var(--danger);">Please paste your GAS Web App URL in the HTML file first!</td></tr>`;
                return;
            }

            fetch(GAS_API_URL)
                .then(response => response.json())
                .then(result => {
                    if(result.status === 'success') {
                        etsList = result.data;
                        document.getElementById('searchInput').value = '';
                        populateLocationFilter();
                        renderTable(etsList);
                        if(showAnim) showToast('Data refreshed successfully');
                    } else {
                        throw new Error(result.message);
                    }
                })
                .catch(error => {
                    showToast('Failed to load data. Check URL or connection.');
                    document.getElementById('etsTableBody').innerHTML = `<tr><td colspan="3" style="text-align:center;padding:30px;color:var(--danger);">Error loading data.</td></tr>`;
                });
        }

        function populateLocationFilter() {
            const select = document.getElementById('locationFilter');
            if (!select) return;
            const uniquePlants = [...new Set(etsList.map(item => {
                if(item.name.includes(' ')) return item.name.split(' ')[0];
                return item.name;
            }))].sort();
            
            let html = `<option value="ALL">All Plants</option>`;
            uniquePlants.forEach(p => {
                html += `<option value="${p}">${p}</option>`;
            });
            select.innerHTML = html;
        }

        function toggleDarkMode() {
            document.body.classList.toggle('dark-mode');
            const isDark = document.body.classList.contains('dark-mode');
            localStorage.setItem('tabreed_theme', isDark ? 'dark' : 'light');
            
            const btn = document.getElementById('darkModeToggleBtn');
            btn.classList.add('pop-anim');
            setTimeout(() => btn.classList.remove('pop-anim'), 350);
        }

        function renderTable(data) {
            const tbody = document.getElementById('etsTableBody');
            
            if (data.length === 0) {
                tbody.innerHTML = `<tr><td colspan="3" style="text-align:center;padding:30px;color:var(--text-muted);">No ETS locations found.</td></tr>`;
                return;
            }

            tbody.innerHTML = data.map((item, idx) => {
                const commentText = (item.comment || '').trim();
                const hasComment = commentText.length > 0;
                
                const noteBadge = hasComment 
                    ? `<span class="note-badge" onclick="showComment('${escapeHtml(commentText)}')"><i class="fas fa-sticky-note"></i> Note</span>` 
                    : '';
                
                const plantName = item.name.includes(' ') ? item.name.split(' ')[0] : item.name;

                return `<tr>
                    <td style="font-weight:700;color:var(--primary);">${escapeHtml(item.name)} ${noteBadge}</td>
                    <td>
                        <div style="font-weight:600; color:var(--text-primary); margin-bottom:4px;">${escapeHtml(plantName)}</div>
                        <span class="location-tag"><i class="fas fa-map-marker-alt"></i> ${item.lat.toFixed(4)}, ${item.lng.toFixed(4)}</span>
                    </td>
                    <td style="text-align:right;">
                        <a href="https://www.google.com/maps/dir/?api=1&destination=${item.lat},${item.lng}" target="_blank" class="btn-dir">
                            <i class="fas fa-directions"></i> <span class="hide-mobile">Direction</span>
                        </a>
                        <button onclick="openUpdateModal(${idx})" class="btn-edit" style="margin-left:5px;">
                            <i class="fas fa-edit"></i>
                        </button>
                    </td>
                </tr>`;
            }).join('');
            
            const tableEl = document.querySelector('.card-container');
            tableEl.classList.remove('subtle-pop-anim');
            void tableEl.offsetWidth;
            tableEl.classList.add('subtle-pop-anim');
        }

        function filterETS() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const locationSelect = document.getElementById('locationFilter');
            const locFilter = locationSelect ? locationSelect.value : 'ALL';
            
            const labelEl = document.getElementById('label-location');
            if (labelEl && locationSelect && locationSelect.options.length > 0) {
                labelEl.innerText = locationSelect.options[locationSelect.selectedIndex].text;
            }

            const filtered = etsList.filter(item => {
                const matchesQuery = item.name.toLowerCase().includes(query) || 
                                     item.lat.toString().includes(query) || 
                                     item.lng.toString().includes(query);
                const matchesLoc = (locFilter === 'ALL' || item.name.includes(locFilter));
                return matchesQuery && matchesLoc;
            });
            
            renderTable(filtered);
            
            const btn = document.getElementById('btn-location-wrapper');
            if (btn && locFilter !== 'ALL') {
                btn.classList.add('filter-active');
                btn.classList.add('pop-anim');
                setTimeout(() => btn.classList.remove('pop-anim'), 350);
            } else if (btn) {
                btn.classList.remove('filter-active');
            }
        }

        function openAddModal() {
            document.getElementById('modalTitle').innerHTML = '<i class="fas fa-map-marked-alt"></i> Add New ETS';
            document.getElementById('editIndex').value = '-1';
            document.getElementById('etsName').value = '';
            document.getElementById('etsLat').value = '';
            document.getElementById('etsLng').value = '';
            document.getElementById('etsComment').value = '';
            document.getElementById('saveBtn').innerHTML = '<i class="fas fa-save"></i> Save ETS';
            document.getElementById('etsModal').style.display = 'flex';
        }

        function openUpdateModal(index) {
            const item = etsList[index];
            document.getElementById('modalTitle').innerHTML = '<i class="fas fa-edit"></i> Edit ETS Location';
            document.getElementById('editIndex').value = index;
            document.getElementById('etsName').value = item.name;
            document.getElementById('etsLat').value = item.lat;
            document.getElementById('etsLng').value = item.lng;
            document.getElementById('etsComment').value = item.comment || '';
            document.getElementById('saveBtn').innerHTML = '<i class="fas fa-check"></i> Update ETS';
            document.getElementById('etsModal').style.display = 'flex';
        }

        function closeModal() {
            document.getElementById('etsModal').style.display = 'none';
        }

        function saveETS() {
            const idx = parseInt(document.getElementById('editIndex').value);
            const name = document.getElementById('etsName').value.trim();
            const lat = parseFloat(document.getElementById('etsLat').value);
            const lng = parseFloat(document.getElementById('etsLng').value);
            const comment = document.getElementById('etsComment').value.trim();

            if (!name || isNaN(lat) || isNaN(lng)) {
                showToast("Please fill Name, Latitude, and Longitude.");
                return;
            }
            
            if(GAS_API_URL === "YOUR_WEB_APP_URL_HERE") {
                showToast("Please configure Web App URL first!");
                return;
            }

            const payload = { action: idx >= 0 ? 'update' : 'add', data: { name, lat, lng, comment } };
            
            document.getElementById('saveBtn').innerHTML = '<i class="fas fa-spinner fa-spin"></i> Saving...';
            document.getElementById('saveBtn').disabled = true;

            fetch(GAS_API_URL, {
                method: "POST",
                body: JSON.stringify(payload)
            })
            .then(res => res.json())
            .then(result => {
                if(result.status === 'success') {
                    showToast(idx >= 0 ? "ETS Updated Successfully!" : "New ETS Added!");
                    closeModal();
                    refreshTable(false); // reload data
                } else {
                    throw new Error(result.message);
                }
            })
            .catch(err => {
                showToast("Error saving data: " + err.message);
            })
            .finally(() => {
                document.getElementById('saveBtn').disabled = false;
                document.getElementById('saveBtn').innerHTML = '<i class="fas fa-save"></i> Save ETS';
            });
        }

        function showComment(text) {
            document.getElementById('commentTextDisplay').innerText = text;
            document.getElementById('commentModal').style.display = 'flex';
        }

        function closeCommentModal() {
            document.getElementById('commentModal').style.display = 'none';
        }

        function showToast(msg) {
            const t = document.getElementById('toastMsg');
            t.innerText = msg;
            t.classList.add('show');
            setTimeout(() => { t.classList.remove('show'); }, 3000);
        }

        function escapeHtml(str) {
            return str.replace(/[&<>'"]/g, 
                tag => ({
                    '&': '&amp;',
                    '<': '&lt;',
                    '>': '&gt;',
                    "'": '&#39;',
                    '"': '&quot;'
                }[tag] || tag)
            );
        }
