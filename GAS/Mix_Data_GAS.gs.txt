// Google Drive PWA.json File ID (Default configuration backup)
const PWA_DRIVE_FILE_ID = '1AlvVbRj3DOQIMOQ2DaWikRoOlilJMmlX';

// Helper: Response Formatter with full CORS Support
function jsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ─────────────────────────────────────────────────────────────
// 1. LOGIN LOG SHEET HANDLER (Simple Login Data with Time Stamp)
// ─────────────────────────────────────────────────────────────
function getLoginLogSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Login Log');
  if (!sheet) {
    sheet = ss.insertSheet('Login Log');
    
    // Header row: Simple, clean login data with timestamp
    var headers = ['Timestamp', 'Employee Code', 'User Name', 'Role', 'Plant / Shift', 'Device / Platform'];
    sheet.appendRow(headers);
    
    // Header formatting
    sheet.setFrozenRows(1);
    var headerRange = sheet.getRange(1, 1, 1, 6);
    headerRange.setFontWeight('bold')
               .setBackground('#0f172a')
               .setFontColor('#ffffff')
               .setHorizontalAlignment('center');
    
    // Set clean, readable column widths
    sheet.setColumnWidth(1, 175); // Timestamp (DD.MM.YYYY HH:mm:ss)
    sheet.setColumnWidth(2, 130); // Employee Code
    sheet.setColumnWidth(3, 170); // User Name
    sheet.setColumnWidth(4, 130); // Role
    sheet.setColumnWidth(5, 140); // Plant / Shift
    sheet.setColumnWidth(6, 190); // Device / Platform

    // Set first column as plain text to preserve full date & time formatting
    sheet.getRange("A:A").setNumberFormat("@");
  }
  return sheet;
}

// Standalone runner function to create or verify "Login Log" sheet in 1-click
function createLoginLogSheet() {
  var sheet = getLoginLogSheet();
  SpreadsheetApp.getActiveSpreadsheet().setActiveSheet(sheet);
  Logger.log('Login Log sheet ready with ' + sheet.getLastRow() + ' rows.');
}

// ─────────────────────────────────────────────────────────────
// 1B. PETTY CASH SHEET HANDLER (Petty Cash Expenses & Claims)
// ─────────────────────────────────────────────────────────────
function getPettyCashSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Petty Cash') || ss.getSheetByName('Pitty Cash');
  if (!sheet) {
    sheet = ss.insertSheet('Petty Cash');
    
    // Header row: Voucher ID, Date of Purchase, Amount (AED), Item / Note, Employee Name, Employee Code, Receipt URL, Status, Remarks, Timestamp
    var headers = ['Voucher ID', 'Date of Purchase', 'Amount (AED)', 'Item / Note', 'Employee Name', 'Employee Code', 'Receipt URL', 'Status', 'Remarks', 'Timestamp'];
    sheet.appendRow(headers);
    
    // Header formatting
    sheet.setFrozenRows(1);
    var headerRange = sheet.getRange(1, 1, 1, 10);
    headerRange.setFontWeight('bold')
               .setBackground('#065f46') // Emerald Green
               .setFontColor('#ffffff')
               .setHorizontalAlignment('center');
    
    // Column widths
    sheet.setColumnWidth(1, 150); // Voucher ID
    sheet.setColumnWidth(2, 140); // Date of Purchase
    sheet.setColumnWidth(3, 120); // Amount (AED)
    sheet.setColumnWidth(4, 250); // Item / Note
    sheet.setColumnWidth(5, 170); // Employee Name
    sheet.setColumnWidth(6, 120); // Employee Code
    sheet.setColumnWidth(7, 200); // Receipt URL / Bill
    sheet.setColumnWidth(8, 120); // Status
    sheet.setColumnWidth(9, 220); // Remarks
    sheet.setColumnWidth(10, 175); // Timestamp

    sheet.getRange("A:A").setNumberFormat("@");
    sheet.getRange("B:B").setNumberFormat("@");
    sheet.getRange("F:F").setNumberFormat("@");
    sheet.getRange("J:J").setNumberFormat("@");
  }
  return sheet;
}

// Standalone runner function for Petty Cash
function createPettyCashSheet() {
  var sheet = getPettyCashSheet();
  SpreadsheetApp.getActiveSpreadsheet().setActiveSheet(sheet);
  Logger.log('Petty Cash sheet ready with ' + sheet.getLastRow() + ' rows.');
}

// Custom Menu in Google Sheets UI
function onOpen() {
  try {
    SpreadsheetApp.getUi()
      .createMenu('Tabreed Mix Data')
      .addItem("Create / Check 'Login Log' Sheet", "createLoginLogSheet")
      .addItem("Create / Check 'Petty Cash' Sheet", "createPettyCashSheet")
      .addItem("Create / Check 'PWA_Sync_Log' Sheet", "getLogSheet")
      .addToUi();
  } catch(e) {}
}

// Helper: Get or Create Sheet Tab for PWA Configuration Sync Logs
function getLogSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('PWA_Sync_Log');
  if (!sheet) {
    sheet = ss.insertSheet('PWA_Sync_Log');
    sheet.appendRow(['Timestamp', 'User', 'Action', 'Status', 'Summary']);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, 5).setFontWeight('bold').setBackground('#1e293b').setFontColor('#ffffff');
    sheet.setColumnWidth(1, 175);
    sheet.setColumnWidth(2, 140);
    sheet.setColumnWidth(3, 130);
    sheet.setColumnWidth(4, 100);
    sheet.setColumnWidth(5, 300);
  }
  return sheet;
}

// ─────────────────────────────────────────────────────────────
// 2. GET REQUESTS: Ping, Read Logs, Setup, or Read PWA Config
// ─────────────────────────────────────────────────────────────
function doGet(e) {
  try {
    var param = (e && e.parameter) ? e.parameter : {};
    var action = param.action || 'getPwaConfig';
    var targetFileId = param.fileId || PWA_DRIVE_FILE_ID;

    // Ping / Health Check
    if (action === 'ping' || action === 'test') {
      return jsonResponse({
        status: 'success',
        message: 'Mix Data GAS is active and connected!',
        fileId: targetFileId,
        timestamp: new Date().toISOString()
      });
    }

    // Direct Create / Verify "Login Log" Sheet Tab
    if (action === 'createLoginLogSheet' || action === 'setup' || action === 'setupLoginLog') {
      var lSheet = getLoginLogSheet();
      return jsonResponse({
        status: 'success',
        message: '"Login Log" sheet is ready in Mix Data spreadsheet!',
        sheet: 'Login Log',
        totalRows: lSheet.getLastRow()
      });
    }

    // Log Login via GET (Zero-CORS fast logging fallback)
    if (action === 'logLogin' || action === 'loginLog') {
      var logSheet = getLoginLogSheet();
      var nowStr = Utilities.formatDate(new Date(), 'GMT+4', 'dd.MM.yyyy HH:mm:ss');
      var code = String(param.code || param.userCode || param.empCode || '').trim();
      var name = String(param.name || param.userName || param.short || param.user || 'User').trim();
      var role = String(param.role || 'User').trim();
      var plantShift = String(param.plantShift || param.plant || param.shift || '-').trim();
      var device = String(param.device || param.platform || 'PWA Web App').trim();
      var timestamp = param.timestamp || nowStr;

      logSheet.appendRow([
        timestamp,
        code,
        name,
        role,
        plantShift,
        device
      ]);

      return jsonResponse({
        status: 'success',
        message: 'Login logged successfully in "Login Log" sheet',
        sheet: 'Login Log',
        timestamp: timestamp,
        user: name,
        code: code
      });
    }

    // Read recent Login Logs (Last 100 entries)
    if (action === 'getLoginLogs') {
      var lSheet = getLoginLogSheet();
      var lastRow = lSheet.getLastRow();
      var logs = [];
      if (lastRow > 1) {
        var startRow = Math.max(2, lastRow - 99);
        var count = lastRow - startRow + 1;
        var dataRange = lSheet.getRange(startRow, 1, count, 6).getValues();
        for (var i = dataRange.length - 1; i >= 0; i--) {
          logs.push({
            timestamp: dataRange[i][0],
            code: dataRange[i][1],
            name: dataRange[i][2],
            role: dataRange[i][3],
            plantShift: dataRange[i][4],
            device: dataRange[i][5]
          });
        }
      }
      return jsonResponse({
        status: 'success',
        sheet: 'Login Log',
        count: logs.length,
        logs: logs
      });
    }

    // Read PWA.json from Google Drive
    if (action === 'getPwaConfig' || action === 'read') {
      var file = DriveApp.getFileById(targetFileId);
      var blob = file.getBlob();
      var text = blob.getDataAsString();
      var jsonData = {};
      try {
        jsonData = text ? JSON.parse(text) : {};
      } catch(parseErr) {
        jsonData = { raw: text };
      }

      return jsonResponse({
        status: 'success',
        source: 'google_drive_pwa_json',
        fileId: targetFileId,
        fileName: file.getName(),
        lastUpdated: file.getLastUpdated().toISOString(),
        data: jsonData
      });
    }

    // Read Petty Cash Vouchers from "Petty Cash" sheet
    if (action === 'getPettyCash' || action === 'getPittyCash') {
      var pSheet = getPettyCashSheet();
      var lastRow = pSheet.getLastRow();
      var vouchers = [];
      if (lastRow > 1) {
        var dataRange = pSheet.getRange(2, 1, lastRow - 1, 10).getValues();
        for (var i = dataRange.length - 1; i >= 0; i--) {
          var row = dataRange[i];
          if (!row[0] && !row[2] && !row[3]) continue; // Skip blank rows
          vouchers.push({
            id: String(row[0] || ''),
            voucherId: String(row[0] || ''),
            date: String(row[1] || ''),
            dateOfPurchase: String(row[1] || ''),
            amount: parseFloat(row[2]) || 0,
            item: String(row[3] || ''),
            description: String(row[3] || ''),
            employeeName: String(row[4] || ''),
            userName: String(row[4] || ''),
            user: String(row[4] || ''),
            employeeCode: String(row[5] || ''),
            empCode: String(row[5] || ''),
            receiptUrl: String(row[6] || ''),
            status: String(row[7] || 'Submitted'),
            remarks: String(row[8] || ''),
            timestamp: String(row[9] || '')
          });
        }
      }
      return jsonResponse({
        status: 'success',
        sheet: 'Petty Cash',
        count: vouchers.length,
        data: vouchers
      });
    }

    // Default Fallback
    return jsonResponse({
      status: 'success',
      message: 'Mix Data Web App is running. Supported actions: logLogin, getLoginLogs, createLoginLogSheet, getPettyCash, addPettyCash, getPwaConfig.',
      targetFileId: targetFileId
    });

  } catch (err) {
    return jsonResponse({
      status: 'error',
      message: 'Mix Data GET error: ' + err.toString()
    });
  }
}

// ─────────────────────────────────────────────────────────────
// 3. POST REQUESTS: Log Login & Save PWA Preferences
// ─────────────────────────────────────────────────────────────
function doPost(e) {
  try {
    var requestData = {};
    if (e && e.postData && e.postData.contents) {
      try {
        requestData = JSON.parse(e.postData.contents);
      } catch (parseErr) {
        requestData = { payload: e.postData.contents };
      }
    }
    if (e && e.parameter) {
      for (var k in e.parameter) {
        if (!requestData[k]) requestData[k] = e.parameter[k];
      }
    }

    var action = requestData.action || 'savePwaConfig';
    var targetFileId = requestData.fileId || PWA_DRIVE_FILE_ID;
    var user = requestData.user || requestData.name || 'Unknown User';

    // 1. ACTION: Log User Login to "Login Log" Sheet
    if (action === 'logLogin' || action === 'loginLog') {
      try {
        var logSheet = getLoginLogSheet();
        var nowStr = Utilities.formatDate(new Date(), 'GMT+4', 'dd.MM.yyyy HH:mm:ss');
        var code = String(requestData.code || requestData.userCode || requestData.empCode || '').trim();
        var name = String(requestData.name || requestData.userName || requestData.short || user || 'User').trim();
        var role = String(requestData.role || 'User').trim();
        var plantShift = String(requestData.plantShift || requestData.plant || requestData.shift || '-').trim();
        var device = String(requestData.device || requestData.platform || 'PWA Web App').trim();
        var timestamp = requestData.timestamp || nowStr;

        logSheet.appendRow([
          timestamp,
          code,
          name,
          role,
          plantShift,
          device
        ]);

        return jsonResponse({
          status: 'success',
          message: 'Login logged successfully in "Login Log" sheet',
          sheet: 'Login Log',
          timestamp: timestamp,
          user: name,
          code: code
        });
      } catch(logErr) {
        return jsonResponse({
          status: 'error',
          message: 'Error writing to Login Log sheet: ' + logErr.toString()
        });
      }
    }

    // 2. ACTION: Save PWA Config to Google Drive PWA.json
    if (action === 'savePwaConfig' || action === 'update' || action === 'save') {
      var pwaPayload = requestData.pwaData || requestData.data || requestData;

      var jsonString = (typeof pwaPayload === 'string') 
        ? pwaPayload 
        : JSON.stringify(pwaPayload, null, 2);

      var driveFile = DriveApp.getFileById(targetFileId);
      driveFile.setContent(jsonString);

      // Log clean 1-line backup entry in PWA_Sync_Log
      try {
        var pwaLogSheet = getLogSheet();
        var nowStr = Utilities.formatDate(new Date(), 'GMT+4', 'dd.MM.yyyy HH:mm:ss');
        var ordersCount = (pwaPayload.moduleOrders && typeof pwaPayload.moduleOrders === 'object')
          ? Object.keys(pwaPayload.moduleOrders).length
          : 0;
        var counter = pwaPayload.syncCounter || 1;
        var summary = 'PWA Config synced (v' + counter + ') for ' + user + ' (' + ordersCount + ' custom orders)';

        pwaLogSheet.appendRow([
          nowStr,
          user,
          action,
          'SUCCESS',
          summary
        ]);

        // Keep log compact (retain last 100 entries)
        if (pwaLogSheet.getLastRow() > 105) {
          pwaLogSheet.deleteRows(2, pwaLogSheet.getLastRow() - 100);
        }
      } catch(pwaLogErr) {}

      return jsonResponse({
        status: 'success',
        message: 'Successfully updated Google Drive PWA.json!',
        fileId: targetFileId,
        fileName: driveFile.getName(),
        updatedAt: new Date().toISOString(),
        syncCounter: pwaPayload.syncCounter || 1,
        user: user
      });
    }

    // 3. ACTION: Add Petty Cash Voucher
    if (action === 'addPettyCash' || action === 'createPettyCash') {
      try {
        var pSheet = getPettyCashSheet();
        var nowStr = Utilities.formatDate(new Date(), 'GMT+4', 'dd.MM.yyyy HH:mm:ss');
        var todayStr = Utilities.formatDate(new Date(), 'GMT+4', 'dd.MM.yyyy');

        var vId = String(requestData.id || requestData.voucherId || ('PCV-' + Date.now().toString().slice(-6))).trim();
        var pDate = String(requestData.dateOfPurchase || requestData.date || todayStr).trim();
        var pAmount = parseFloat(requestData.amount || requestData.cashPaid || 0) || 0;
        var pItem = String(requestData.item || requestData.note || requestData.description || '').trim();
        var pEmpName = String(requestData.employeeName || requestData.userName || requestData.user || user || '').trim();
        var pEmpCode = String(requestData.employeeCode || requestData.empCode || requestData.code || '').trim();
        var pReceipt = String(requestData.receiptUrl || requestData.billUrl || requestData.bill || '').trim();
        var pStatus = String(requestData.status || 'Submitted').trim();
        var pRemarks = String(requestData.remarks || '').trim();
        var pTimestamp = requestData.timestamp || nowStr;

        pSheet.appendRow([
          vId,
          pDate,
          pAmount,
          pItem,
          pEmpName,
          pEmpCode,
          pReceipt,
          pStatus,
          pRemarks,
          pTimestamp
        ]);

        return jsonResponse({
          status: 'success',
          message: 'Petty cash voucher added successfully to Google Sheet',
          voucherId: vId,
          data: {
            id: vId,
            voucherId: vId,
            dateOfPurchase: pDate,
            date: pDate,
            amount: pAmount,
            item: pItem,
            description: pItem,
            employeeName: pEmpName,
            userName: pEmpName,
            user: pEmpName,
            employeeCode: pEmpCode,
            empCode: pEmpCode,
            receiptUrl: pReceipt,
            status: pStatus,
            remarks: pRemarks,
            timestamp: pTimestamp
          }
        });
      } catch(addErr) {
        return jsonResponse({
          status: 'error',
          message: 'Error adding to Petty Cash sheet: ' + addErr.toString()
        });
      }
    }

    // 4. ACTION: Update Petty Cash Voucher (Status / Remarks / Edit)
    if (action === 'updatePettyCash') {
      try {
        var pSheet = getPettyCashSheet();
        var targetId = String(requestData.id || requestData.voucherId || '').trim();
        var lastRow = pSheet.getLastRow();
        var foundRow = -1;

        if (targetId && lastRow > 1) {
          var idColumn = pSheet.getRange(2, 1, lastRow - 1, 1).getValues();
          for (var r = 0; r < idColumn.length; r++) {
            if (String(idColumn[r][0]).trim() === targetId) {
              foundRow = r + 2;
              break;
            }
          }
        }

        if (foundRow > 1) {
          if (requestData.dateOfPurchase || requestData.date) pSheet.getRange(foundRow, 2).setValue(String(requestData.dateOfPurchase || requestData.date).trim());
          if (requestData.amount !== undefined) pSheet.getRange(foundRow, 3).setValue(parseFloat(requestData.amount) || 0);
          if (requestData.item !== undefined || requestData.description !== undefined) pSheet.getRange(foundRow, 4).setValue(String(requestData.item || requestData.description || '').trim());
          if (requestData.receiptUrl !== undefined) pSheet.getRange(foundRow, 7).setValue(String(requestData.receiptUrl || '').trim());
          if (requestData.status !== undefined) pSheet.getRange(foundRow, 8).setValue(String(requestData.status || '').trim());
          if (requestData.remarks !== undefined) pSheet.getRange(foundRow, 9).setValue(String(requestData.remarks || '').trim());

          return jsonResponse({
            status: 'success',
            message: 'Petty cash voucher updated in Google Sheet',
            voucherId: targetId,
            row: foundRow
          });
        }

        return jsonResponse({
          status: 'error',
          message: 'Petty cash voucher with ID ' + targetId + ' not found in Google Sheet'
        });
      } catch(upErr) {
        return jsonResponse({
          status: 'error',
          message: 'Error updating Petty Cash sheet: ' + upErr.toString()
        });
      }
    }

    // 5. ACTION: Delete Petty Cash Voucher
    if (action === 'deletePettyCash') {
      try {
        var pSheet = getPettyCashSheet();
        var targetId = String(requestData.id || requestData.voucherId || '').trim();
        var lastRow = pSheet.getLastRow();
        var foundRow = -1;

        if (targetId && lastRow > 1) {
          var idColumn = pSheet.getRange(2, 1, lastRow - 1, 1).getValues();
          for (var r = 0; r < idColumn.length; r++) {
            if (String(idColumn[r][0]).trim() === targetId) {
              foundRow = r + 2;
              break;
            }
          }
        }

        if (foundRow > 1) {
          pSheet.deleteRow(foundRow);
          return jsonResponse({
            status: 'success',
            message: 'Petty cash voucher deleted from Google Sheet',
            voucherId: targetId
          });
        }

        return jsonResponse({
          status: 'error',
          message: 'Voucher ID ' + targetId + ' not found in Google Sheet'
        });
      } catch(delErr) {
        return jsonResponse({
          status: 'error',
          message: 'Error deleting from Petty Cash sheet: ' + delErr.toString()
        });
      }
    }

    return jsonResponse({
      status: 'error',
      message: 'Unrecognized action: ' + action
    });

  } catch (err) {
    return jsonResponse({
      status: 'error',
      message: 'Mix Data GAS error: ' + err.toString()
    });
  }
}
