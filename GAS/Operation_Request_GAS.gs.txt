function onOpen() {
  try {
    SpreadsheetApp.getUi()
      .createMenu('Tabreed Automation')
      .addItem('🧹 Clean Empty / Ghost Rows', 'cleanInvalidRows')
      .addItem('👥 Populate Team Contacts Sheet', 'populateTeamContactsSheet')
      .addItem('📜 Setup History Sheet', 'setupHistorySheet')
      .addItem('🔄 Run Full System Setup', 'setupInitialSheets')
      .addToUi();
  } catch(e) {
    Logger.log("onOpen UI error: " + e.toString());
  }
}

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  var s = ss.getSheetByName("Operation Request");
  if (s) return s;
  s = ss.getSheetByName("Sheet1");
  if (s) return s;
  return ss.getSheets()[0];
}

/* 🧹 Clean all empty or ghost rows from the Operation Request sheet & fix corrupted IDs */
function cleanInvalidRows() {
  var sheet = getSheet();
  var allData = sheet.getDataRange().getValues();
  var deletedCount = 0;
  
  // 1. First pass: delete ghost/empty rows backwards so row indices don't shift
  for (var i = allData.length - 1; i >= 1; i--) {
    var row = allData[i];
    var wo = String(row[0] || '').trim();
    var desc = String(row[1] || '').trim();
    var plant = String(row[2] || '').trim();
    
    // If Work Order, Description, and Plant are all empty, delete this ghost row
    if (!wo && !desc && !plant) {
      sheet.deleteRow(i + 1);
      deletedCount++;
    }
  }

  // 2. Second pass: Clear Column I and any extra columns beyond H (no unnecessary ID column in Sheet)
  if (sheet.getMaxColumns() >= 9) {
    sheet.getRange(1, 9, sheet.getMaxRows(), sheet.getMaxColumns() - 8).clearContent();
  }

  return deletedCount;
}

/* 🧹 Helper to clear Column I (ID) directly */
function removeColumnI_ID() {
  var sheet = getSheet();
  if (!sheet) return;
  if (sheet.getMaxColumns() >= 9) {
    sheet.getRange(1, 9, sheet.getMaxRows(), sheet.getMaxColumns() - 8).clearContent();
  }
}

/* 🔄 Complete initial setup: cleans ghost rows and ensures History + Team Contacts are active */
function setupInitialSheets() {
  var cleaned = cleanInvalidRows();
  populateTeamContactsSheet();
  setupHistorySheet();
  return { status: "success", message: "Setup completed! Cleaned " + cleaned + " empty rows. Team Contacts and History sheets ready.", cleanedRows: cleaned };
}

function logToHistorySheet(action, wo, plant, desc, user, details) {
  try {
    var hSheet = getHistorySheet();
    var now = new Date();
    var timestamp = Utilities.formatDate(now, "GMT+4", "yyyy-MM-dd'T'HH:mm:ss'Z'");
    var displayTime = Utilities.formatDate(now, "GMT+4", "dd.MM.yyyy HH:mm");
    hSheet.appendRow([
      timestamp,
      displayTime,
      action || '',
      wo || '',
      plant || '',
      desc || '',
      user || 'Technician',
      details || ''
    ]);
  } catch(e) {
    Logger.log("Failed to write to History sheet: " + e.toString());
  }
}

function doGet(e) {
  try {
    var paramAction = (e && e.parameter && e.parameter.action) ? String(e.parameter.action) : '';

    // 1. Clean Ghost / Invalid Rows via GET
    if (paramAction === 'cleanEmptyRows' || paramAction === 'cleanKachra') {
      var cleaned = cleanInvalidRows();
      return jsonResponse({ status: "success", message: "Cleaned " + cleaned + " ghost rows", cleanedCount: cleaned });
    }

    // 2. Full Setup via GET
    if (paramAction === 'setupAll' || paramAction === 'setupInitialSheets') {
      var res = setupInitialSheets();
      return jsonResponse(res);
    }

    // 3. Fetch History via GET
    if (paramAction === 'getHistory') {
      var hSheet = getHistorySheet();
      var hData = hSheet.getDataRange().getValues();
      var histList = [];
      for (var hi = 1; hi < hData.length; hi++) {
        var hRow = hData[hi];
        if (!hRow[0] && !hRow[2] && !hRow[5]) continue;
        histList.unshift({
          id: 'cloud_hist_' + hi,
          timestamp: hRow[0] instanceof Date ? hRow[0].toISOString() : String(hRow[0]),
          displayTime: hRow[1] instanceof Date ? Utilities.formatDate(hRow[1], "GMT+4", "dd.MM.yyyy HH:mm") : String(hRow[1]),
          action: String(hRow[2] || 'info'),
          workOrder: String(hRow[3] || ''),
          plant: String(hRow[4] || ''),
          description: String(hRow[5] || ''),
          user: String(hRow[6] || 'Technician'),
          details: String(hRow[7] || '')
        });
      }
      return jsonResponse({ status: "success", data: histList });
    }

    // 4. Fetch Team Contacts via GET
    if (paramAction === 'getTeamContacts') {
      var tcSheet = getTeamContactsSheet();
      var tcData = tcSheet.getDataRange().getValues();
      var contactsList = [];
      for (var ci = 1; ci < tcData.length; ci++) {
        var row = tcData[ci];
        var code = row[0] ? String(row[0]).trim() : '';
        if (!code) continue;
        contactsList.push({
          code: code,
          short: row[1] ? String(row[1]).trim() : '',
          full: row[2] ? String(row[2]).trim() : '',
          role: row[3] ? String(row[3]).trim() : '',
          phone: row[4] ? String(row[4]).trim() : '',
          whatsapp: row[5] ? String(row[5]).trim() : '',
          email: row[6] ? String(row[6]).trim() : '',
          lastUpdated: row[7] ? String(row[7]).trim() : ''
        });
      }
      return jsonResponse({ status: "success", data: contactsList });
    }

    // 5. Clear History via GET
    if (paramAction === 'clearHistory') {
      var hSheet = getHistorySheet();
      var lastRow = hSheet.getLastRow();
      if (lastRow > 1) {
        hSheet.deleteRows(2, lastRow - 1);
      }
      return jsonResponse({ status: "success", message: "History cleared from sheet" });
    }

    // 6. Fetch PWA Config from Google Drive PWA.json via GET
    if (paramAction === 'getPwaConfig') {
      var targetFileId = (e && e.parameter && e.parameter.fileId) ? String(e.parameter.fileId) : '1AlvVbRj3DOQIMOQ2DaWikRoOlilJMmlX';
      try {
        var pwaFile = DriveApp.getFileById(targetFileId);
        var pwaText = pwaFile.getBlob().getDataAsString();
        var pwaJson = pwaText ? JSON.parse(pwaText) : {};
        return jsonResponse({ status: "success", data: pwaJson, fileId: targetFileId });
      } catch(dErr) {
        return jsonResponse({ status: "error", message: "Error reading Drive file: " + dErr.toString() });
      }
    }

    const sheet = getSheet();
    
    // Check if another action was passed via GET query parameter
    if (paramAction) {
      return handleAction(e.parameter, sheet);
    }

    const allData = sheet.getDataRange().getValues();
    if (allData.length <= 1) {
      return jsonResponse({ status: "success", data: [] });
    }
    
    const rows = [];
    
    for (var i = 1; i < allData.length; i++) {
      var r = allData[i];
      var wo = r[0] ? String(r[0]).trim() : '';
      var desc = r[1] ? String(r[1]).trim() : '';
      var plant = r[2] ? String(r[2]).trim() : '';
      
      // CRITICAL: Skip empty ghost rows so they are never returned to the app
      if (!wo && !desc && !plant) continue;
      
      // CRITICAL FIX: Only Column I (index 8) is the ID. NEVER use r[7] (Remarks) as an ID!
      var stableId = (r[8] && String(r[8]).trim() && String(r[8]).trim().toLowerCase() !== 'urgent') 
        ? String(r[8]).trim() 
        : ("row_" + (i + 1) + "_" + (r[0] ? String(r[0]).trim() : Date.now()));
      
      rows.push({
        "rowIdx": i + 1,
        "Work Order": wo,
        "Description": desc,
        "Plant": plant,
        "Date": r[3] ? (r[3] instanceof Date ? Utilities.formatDate(r[3], "GMT+4", "dd.MM.yyyy") : String(r[3])) : '',
        "Time": r[4] ? String(r[4]) : '',
        "Team": r[5] ? String(r[5]) : '',
        "Status": r[6] ? String(r[6]) : 'Requested',
        "Remarks": r[7] ? String(r[7]) : '',
        "ID": stableId
      });
    }
    
    return jsonResponse({ status: "success", data: rows });
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ status: "error", message: "No data received" });
    }
    
    const data = JSON.parse(e.postData.contents);
    const sheet = getSheet();
    return handleAction(data, sheet);
  } catch (err) {
    return jsonResponse({ status: "error", message: err.toString() });
  }
}

function handleAction(data, sheet) {
  const action = data.action;
  if (!action) {
    return jsonResponse({ status: "error", message: "No action specified" });
  }

  // 0. CLEAN GHOST / EMPTY ROWS
  if (action === 'cleanEmptyRows' || action === 'cleanKachra') {
    var count = cleanInvalidRows();
    return jsonResponse({ status: "success", message: "Cleaned " + count + " invalid ghost rows", count: count });
  }

  // 0B. FULL SETUP ACTION
  if (action === 'setupAll' || action === 'setupInitialSheets') {
    return jsonResponse(setupInitialSheets());
  }

  const allData = sheet.getDataRange().getValues();
  
  // Helper to find target row in sheet (1-based index)
  function findRow() {
    if (data.rowIdx) {
      var rIdx = parseInt(data.rowIdx, 10);
      if (rIdx >= 2 && rIdx <= allData.length) {
        var rVal = allData[rIdx - 1];
        if (data.workOrder && String(rVal[0] || '').trim().toLowerCase() === String(data.workOrder).trim().toLowerCase()) {
          return rIdx;
        }
        if (data.id && (String(rVal[8] || '').trim() === String(data.id).trim() || String(rVal[7] || '').trim() === String(data.id).trim())) {
          return rIdx;
        }
      }
    }
    
    var targetId = data.id ? String(data.id).trim() : '';
    var targetWo = data.workOrder ? String(data.workOrder).trim().toLowerCase() : '';

    for (var i = 1; i < allData.length; i++) {
      var rowIdCol9 = String(allData[i][8] || '').trim();
      var rowIdCol8 = String(allData[i][7] || '').trim();
      var rowWo = String(allData[i][0] || '').trim().toLowerCase();
      
      if (targetId && (rowIdCol9 === targetId || rowIdCol8 === targetId || ("row_" + i) === targetId)) {
        return i + 1;
      }
      if (targetWo && rowWo === targetWo) {
        return i + 1;
      }
    }
    return -1;
  }
  
  // 1. DELETE ACTION
  if (action === 'delete') {
    var targetRow = findRow();
    if (targetRow !== -1) {
      var rowValues = sheet.getRange(targetRow, 1, 1, 8).getValues()[0];
      var targetWo = rowValues[0] || data.workOrder || '';
      var targetPlant = rowValues[2] || data.plant || '';
      sheet.deleteRow(targetRow);
      if (!data.skipAutoHistory && !data.skipHistoryLog) {
        logToHistorySheet('delete', targetWo, targetPlant, 'Deleted Work Order #' + (targetWo || targetRow), data.user || 'Supervisor', data.details || '');
      }
      return jsonResponse({ status: "success", message: "Deleted successfully", deletedRow: targetRow });
    }
    return jsonResponse({ status: "error", message: "ID or Work Order not found for delete" });
  }
  
  // 2. UPDATE STATUS ACTION
  if (action === 'updateStatus') {
    var targetRow = findRow();
    if (targetRow !== -1) {
      var rowValues = sheet.getRange(targetRow, 1, 1, 8).getValues()[0];
      var targetWo = rowValues[0] || data.workOrder || '';
      var targetPlant = rowValues[2] || data.plant || '';
      var oldStatus = rowValues[6] || '';
      sheet.getRange(targetRow, 7).setValue(data.status);
      if (!data.skipAutoHistory && !data.skipHistoryLog) {
        logToHistorySheet('status', targetWo, targetPlant, 'Status changed from ' + oldStatus + ' to ' + data.status, data.user || 'Supervisor', data.details || '');
      }
      return jsonResponse({ status: "success", message: "Status updated", updatedRow: targetRow });
    }
    return jsonResponse({ status: "error", message: "ID or Work Order not found for updateStatus" });
  }
  
  // 3. FULL UPDATE ACTION
  if (action === 'update') {
    var targetRow = findRow();
    if (targetRow !== -1) {
      var currentVals = sheet.getRange(targetRow, 1, 1, 8).getValues()[0];
      var newWO = (data.workOrder !== undefined && data.workOrder !== null && String(data.workOrder).trim() !== '') ? data.workOrder : currentVals[0];
      var newDesc = (data.description !== undefined && data.description !== null && String(data.description).trim() !== '') ? data.description : currentVals[1];
      var newPlant = (data.plant !== undefined && data.plant !== null && String(data.plant).trim() !== '') ? data.plant : currentVals[2];
      var newDate = (data.date !== undefined && data.date !== null && String(data.date).trim() !== '') ? data.date : currentVals[3];
      var newTime = (data.time !== undefined && data.time !== null && String(data.time).trim() !== '') ? data.time : currentVals[4];
      var newTeam = (data.team !== undefined && data.team !== null && String(data.team).trim() !== '') ? data.team : currentVals[5];
      var newStatus = (data.status !== undefined && data.status !== null && String(data.status).trim() !== '') ? data.status : (currentVals[6] || 'Requested');
      var newRemarks = (data.remarks !== undefined && data.remarks !== null) ? data.remarks : currentVals[7];

      // CRITICAL GUARD: Never write an entirely blank row to the sheet
      if (!newWO && !newDesc && !newPlant) {
        return jsonResponse({ status: "error", message: "Update rejected: Empty data would corrupt row" });
      }

      sheet.getRange(targetRow, 1, 1, 8).setValues([[
        newWO,
        newDesc,
        newPlant,
        newDate,
        newTime,
        newTeam,
        newStatus,
        newRemarks
      ]]);
      if (!data.skipAutoHistory && !data.skipHistoryLog) {
        logToHistorySheet('update', newWO, newPlant, newDesc, data.user || 'Technician', data.details || newRemarks || '');
      }
      return jsonResponse({ status: "success", message: "Updated successfully", updatedRow: targetRow });
    }
    return jsonResponse({ status: "error", message: "ID or Work Order not found for update" });
  }
  
  // 4. DIRECT LOG HISTORY ACTION
  if (action === 'logHistory') {
    var h = data.historyItem || data;
    logToHistorySheet(h.action, h.workOrder, h.plant, h.description, h.user, h.details);
    return jsonResponse({ status: "success", message: "History logged to Sheet: History" });
  }

  // 4B. CLEAR HISTORY ACTION
  if (action === 'clearHistory') {
    var hSheet = getHistorySheet();
    var lastRow = hSheet.getLastRow();
    if (lastRow > 1) {
      hSheet.deleteRows(2, lastRow - 1);
    }
    return jsonResponse({ status: "success", message: "History sheet cleared in cloud" });
  }

  // 5. SAVE SINGLE TEAM CONTACT
  if (action === 'saveTeamContact') {
    var tcSheet = getTeamContactsSheet();
    var targetCode = String(data.code || '').trim();
    var tcData = tcSheet.getDataRange().getValues();
    var foundRow = -1;
    
    for (var ti = 1; ti < tcData.length; ti++) {
      if (String(tcData[ti][0]).trim().toLowerCase() === targetCode.toLowerCase()) {
        foundRow = ti + 1;
        break;
      }
    }
    
    var nowStr = Utilities.formatDate(new Date(), "GMT+4", "dd.MM.yyyy HH:mm");
    if (foundRow !== -1) {
      if (data.short) tcSheet.getRange(foundRow, 2).setValue(data.short);
      if (data.full) tcSheet.getRange(foundRow, 3).setValue(data.full);
      if (data.role) tcSheet.getRange(foundRow, 4).setValue(data.role);
      if (data.phone !== undefined) tcSheet.getRange(foundRow, 5).setValue(data.phone);
      if (data.whatsapp !== undefined) tcSheet.getRange(foundRow, 6).setValue(data.whatsapp);
      if (data.email !== undefined) tcSheet.getRange(foundRow, 7).setValue(data.email);
      tcSheet.getRange(foundRow, 8).setValue(nowStr);
    } else {
      tcSheet.appendRow([
        targetCode,
        data.short || '',
        data.full || '',
        data.role || '',
        data.phone || '',
        data.whatsapp || '',
        data.email || '',
        nowStr
      ]);
    }
    
    logToHistorySheet('contact_update', '', '', 'Updated Team Contact #' + targetCode + ' (' + (data.short || data.full || '') + ')', data.user || 'User', 'Phone: ' + (data.phone || 'N/A') + ', WA: ' + (data.whatsapp || 'N/A') + ', Email: ' + (data.email || 'N/A'));
    return jsonResponse({ status: "success", message: "Team Contact saved to Google Sheet", code: targetCode });
  }

  // 5B. GET PWA CONFIG (Read from Google Drive PWA.json)
  if (action === 'getPwaConfig') {
    var gFid = data.fileId || '1AlvVbRj3DOQIMOQ2DaWikRoOlilJMmlX';
    try {
      var dFile = DriveApp.getFileById(gFid);
      var text = dFile.getBlob().getDataAsString();
      return jsonResponse({ status: "success", data: text ? JSON.parse(text) : {} });
    } catch(err) {
      return jsonResponse({ status: "error", message: "Drive read error: " + err.toString() });
    }
  }

  // 5C. SAVE PWA CONFIG (Write directly to Google Drive PWA.json)
  if (action === 'savePwaConfig') {
    var sFid = data.fileId || '1AlvVbRj3DOQIMOQ2DaWikRoOlilJMmlX';
    try {
      var dFile = DriveApp.getFileById(sFid);
      var payload = data.pwaData || data.data || data;
      var fileContent = (typeof payload === 'string') ? payload : JSON.stringify(payload, null, 2);
      dFile.setContent(fileContent);
      return jsonResponse({
        status: "success",
        message: "Saved to Google Drive PWA.json successfully",
        fileId: sFid,
        timestamp: new Date().toISOString()
      });
    } catch(err) {
      return jsonResponse({ status: "error", message: "Drive write error: " + err.toString() });
    }
  }

  // 6. BULK SYNC TEAM CONTACTS (Initial Population or Full Sync)
  if (action === 'bulkSaveTeamContacts' || action === 'syncAllTeamContacts') {
    var tcSheet = getTeamContactsSheet();
    var list = data.contacts || getDefaultTeamContacts();
    var nowStr = Utilities.formatDate(new Date(), "GMT+4", "dd.MM.yyyy HH:mm");
    
    tcSheet.clear();
    tcSheet.appendRow(["Code", "Short Name", "Full Name", "Role", "Tabreed Phone", "WhatsApp", "Email", "Last Updated"]);
    tcSheet.setFrozenRows(1);
    tcSheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#f3f4f6");
    
    for (var bi = 0; bi < list.length; bi++) {
      var c = list[bi];
      tcSheet.appendRow([
        c.code || '',
        c.short || '',
        c.full || '',
        c.role || '',
        c.phone || '',
        c.whatsapp || '',
        c.email || '',
        nowStr
      ]);
    }
    return jsonResponse({ status: "success", message: "All " + list.length + " team contacts synced to sheet" });
  }

  // 7. CREATE ACTION (STRICT VALIDATION & DEDUPLICATION)
  if (action === 'create') {
    var woVal = String(data.workOrder || '').trim();
    var descVal = String(data.description || '').trim();
    var plantVal = String(data.plant || '').trim();

    // STRICT CHECK: Never append empty rows!
    if (!woVal && !descVal && !plantVal) {
      return jsonResponse({ status: "error", message: "Cannot create empty request: Work Order, Description or Plant required" });
    }

    // Deduplication check: inspect the last row to avoid duplicate insertion
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var lastVals = sheet.getRange(lastRow, 1, 1, 3).getValues()[0];
      if (String(lastVals[0]).trim() === woVal && 
          String(lastVals[1]).trim() === descVal && 
          String(lastVals[2]).trim() === plantVal) {
        return jsonResponse({ status: "success", message: "Request already recorded (deduplicated)", id: data.id || ("row_" + lastRow) });
      }
    }

    var newId = data.id || ("row_" + new Date().getTime());
    sheet.appendRow([
      woVal,
      descVal,
      plantVal,
      data.date || '',
      data.time || '',
      data.team || '',
      data.status || 'Requested',
      data.remarks || ''
    ]);
    if (!data.skipAutoHistory && !data.skipHistoryLog) {
      logToHistorySheet('create', woVal, plantVal, descVal, data.user || 'Technician', data.remarks || '');
    }
    
    return jsonResponse({ status: "success", message: "Created successfully", id: newId });
  }

  // SAFE FALLBACK: If action is unrecognized, reject with error — NEVER append garbage!
  return jsonResponse({ status: "error", message: "Unknown action: " + action });
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
